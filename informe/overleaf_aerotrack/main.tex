%--------------------- Marcadores [VERIFICAR] -------------------------
% \verificar{texto} marca una afirmacion que sigue sin confirmar contra
% su fuente. NO se imprime en el PDF: el texto queda en el fuente y cada
% marcador deja rastro en el registro de compilacion, con un recuento
% final ("MARCADORES [VERIFICAR] EN EL CUERPO: n").
% La macro abre con \unskip para no dejar doble espacio cuando el
% marcador va en una linea propia, y no termina en \ignorespaces para no
% comerse el espacio siguiente.
\newcounter{verificarcount}
\newcommand{\verificar}[1]{%
  \ifhmode\unskip\fi
  \stepcounter{verificarcount}%
  \typeout{[VERIFICAR \theverificarcount] marcador oculto en pagina \thepage.}}
\AtEndDocument{%
  \typeout{^^J=====================================================^^J%
    MARCADORES [VERIFICAR] EN EL CUERPO: \theverificarcount^^J%
    =====================================================^^J}}
